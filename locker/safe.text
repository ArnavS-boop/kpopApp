      console.error(`   ${issue.path.join(".")}: ${issue.message}`);

  return result.data;

// ─── STRIPE WEBHOOK (raw body – must precede express.json) ────────────────────
app.post(
  "/webhooks/stripe",
  express.raw({ type: "application/json" }),
  async (req: Request, res: Response): Promise<void> => {
    const sig = req.headers["stripe-signature"];
    if (!sig || typeof sig !== "string") {
      res.status(400).send("Missing stripe-signature header");
      return;
    }

    let event: Stripe.Event;
    try {
      event = stripe.webhooks.constructEvent(req.body as Buffer, sig, env.STRIPE_WEBHOOK_SECRET);
    } catch {
      res.status(400).send("Webhook signature verification failed");
      return;
    }

    if (event.type === "payment_intent.succeeded") {
      const pi        = event.data.object as Stripe.PaymentIntent;
      const orderId   = pi.metadata["orderId"];
      const listingId = pi.metadata["listingId"];

      // Deterministic: use metadata IDs, no join-scan fallback
      if (orderId && listingId) {
        await prisma.$transaction([
          prisma.order.update({
            where: { id: orderId },
            data:  { status: OrderStatus.COMPLETED },
          }),
          prisma.listing.update({
            where: { id: listingId },
            data:  { isSold: true },
          }),
        ]);
      }
    }

    res.json({ received: true });
  }
);

// ─── MULTER ───────────────────────────────────────────────────────────────────
const upload = multer({
  storage: multer.memoryStorage(),
  limits:  { fileSize: 10 * 1024 * 1024 },
});

async function uploadToCloudinary(buffer: Buffer, folder: string): Promise<string> {
  return new Promise((resolve, reject) => {
    const stream = cloudinary.uploader.upload_stream({ folder }, (err, result) => {
      if (err || !result) return reject(err ?? new Error("Cloudinary upload failed"));
      resolve(result.secure_url);
    });
    Readable.from(buffer).pipe(stream);
  });
}

// ─── VALIDATION HELPER ────────────────────────────────────────────────────────
function validate<T>(schema: z.ZodSchema<T>, data: unknown): { data: T } | { error: string } {
  const result = schema.safeParse(data);
  if (!result.success) {
    const msg = result.error.issues
      .map((i) => `${i.path.join(".")}: ${i.message}`)
      .join("; ");
    return { error: msg };
  }
  return { data: result.data };
}

// ─── AUTH TYPES + MIDDLEWARE ──────────────────────────────────────────────────
interface AuthRequest extends Request {
  userId: string;
}

interface AuthSocket extends Socket {
  userId: string;
}

function auth(req: Request, res: Response, next: NextFunction): void {
  const token = req.headers.authorization?.split(" ")[1];
  if (!token) { res.status(401).json({ error: "Unauthorized" }); return; }
  try {
    const payload = jwt.verify(token, env.JWT_SECRET) as JwtPayload;
    if (typeof payload.sub !== "string") throw new Error("Bad token payload");
    (req as AuthRequest).userId = payload.sub;
    next();
  } catch {
    res.status(401).json({ error: "Invalid token" });
  }
}

// ─── AUTH ROUTES ─────────────────────────────────────────────────────────────
app.post("/auth/register", authLimiter, async (req: Request, res: Response): Promise<void> => {
  const parsed = validate(RegisterSchema, req.body);
  if ("error" in parsed) { res.status(400).json({ error: parsed.error }); return; }
  const { username, email, password } = parsed.data;

  const passwordHash = await bcrypt.hash(password, 12);
  try {
    const user  = await prisma.user.create({ data: { username, email, passwordHash } });
    const token = jwt.sign({ sub: user.id }, env.JWT_SECRET, { expiresIn: "7d" });
    res.status(201).json({ token, user: { id: user.id, username: user.username, email: user.email } });
  } catch {
    res.status(409).json({ error: "Username or email already taken" });
  }
});

app.post("/auth/login", authLimiter, async (req: Request, res: Response): Promise<void> => {
  const parsed = validate(LoginSchema, req.body);
  if ("error" in parsed) { res.status(400).json({ error: parsed.error }); return; }
  const { email, password } = parsed.data;

  const user = await prisma.user.findUnique({ where: { email } });
  if (!user || !(await bcrypt.compare(password, user.passwordHash))) {
    res.status(401).json({ error: "Invalid credentials" }); return;
  }
  const token = jwt.sign({ sub: user.id }, env.JWT_SECRET, { expiresIn: "7d" });
  res.json({ token, user: { id: user.id, username: user.username, email: user.email } });
});

app.get("/listings", async (req: Request, res: Response): Promise<void> => {
  const parsed = validate(ListingQuerySchema, req.query);
  if ("error" in parsed) { res.status(400).json({ error: parsed.error }); return; }
  const { groupName, era, condition, minPrice, maxPrice, page, limit } = parsed.data;

  const listings = await prisma.listing.findMany({
    where: {
      isSold: false,
      ...(groupName && { groupName: { contains: groupName, mode: "insensitive" } }),
      ...(era       && { era }),
      ...(condition && { condition }),
      ...((minPrice !== undefined || maxPrice !== undefined) && {
        price: {
          ...(minPrice !== undefined && { gte: minPrice }),
          ...(maxPrice !== undefined && { lte: maxPrice }),
        },
      }),
    },
    include: { images: true, seller: { select: { id: true, username: true, rating: true } } },
    orderBy: { createdAt: "desc" },
    skip:    (page - 1) * limit,
    take:    limit,
  });
  res.json(listings);
});

// ─── ORDER ROUTES ─────────────────────────────────────────────────────────────
const PLATFORM_FEE_PERCENT = 0.1;

app.post("/orders", auth, async (req: Request, res: Response): Promise<void> => {
  const { userId } = req as AuthRequest;
  const parsed = validate(OrderCreateSchema, req.body);
  if ("error" in parsed) { res.status(400).json({ error: parsed.error }); return; }
  const { listingId } = parsed.data;

  const listing = await prisma.listing.findUnique({
    where:   { id: listingId },
    include: { seller: true },
  });
  if (!listing || listing.isSold)   { res.status(400).json({ error: "Listing unavailable" }); return; }
  if (listing.sellerId === userId)  { res.status(400).json({ error: "Cannot buy your own listing" }); return; }

  const platformFee = parseFloat((listing.price * PLATFORM_FEE_PERCENT).toFixed(2));

  // Transaction: persist order first → attach PaymentIntent metadata → update order with PI id
  const { order, clientSecret } = await prisma.$transaction(async (tx) => {
    const newOrder = await tx.order.create({
      data: {
        listingId,
        buyerId:    userId,
        sellerId:   listing.sellerId,
        amount:     listing.price,
        platformFee,
        status:     OrderStatus.PENDING,
      },
    });

    const pi = await stripe.paymentIntents.create({
      amount:                 Math.round(listing.price * 100),
      currency:               "usd",
      application_fee_amount: Math.round(platformFee * 100),
      ...(listing.seller.stripeAccountId && {
        transfer_data: { destination: listing.seller.stripeAccountId },
      }),
      metadata: {
        orderId:   newOrder.id,
        buyerId:   userId,
        listingId,
        sellerId:  listing.sellerId,
      },
    });

    const updatedOrder = await tx.order.update({
      where: { id: newOrder.id },
      data:  { stripePaymentId: pi.id },
    });

    return { order: updatedOrder, clientSecret: pi.client_secret };
  });

  res.status(201).json({ order, clientSecret });
});

// ─── REVIEW ROUTES ────────────────────────────────────────────────────────────
app.post("/reviews", auth, async (req: Request, res: Response): Promise<void> => {
  const { userId } = req as AuthRequest;
  const parsed = validate(ReviewCreateSchema, req.body);
  if ("error" in parsed) { res.status(400).json({ error: parsed.error }); return; }
  const { orderId, rating, comment, proofImage } = parsed.data;

  const order = await prisma.order.findUnique({ where: { id: orderId } });
  if (!order || order.buyerId !== userId)     { res.status(403).json({ error: "Forbidden" }); return; }
  if (order.status !== OrderStatus.COMPLETED) { res.status(400).json({ error: "Order not completed" }); return; }

  // Transaction: create review + recalculate rating atomically (no race between two reviewers)
  const review = await prisma.$transaction(async (tx) => {
    const newReview = await tx.review.create({
      data: {
        orderId,
        reviewerId: userId,
        reviewedId: order.sellerId,
        rating,
        comment,
        proofImage,
      },
    });

    const agg = await tx.review.aggregate({
      where: { reviewedId: order.sellerId },
      _avg:  { rating: true },
    });

    await tx.user.update({
      where: { id: order.sellerId },
      data:  { rating: agg._avg.rating ?? 0 },
    });

    return newReview;
  });

  res.status(201).json(review);
});

// ─── SOCKET.IO ────────────────────────────────────────────────────────────────
const userSockets = new Map<string, Set<string>>();

io.use((socket, next) => {
  const token = socket.handshake.auth?.token as string | undefined;
  if (!token) { next(new Error("Unauthorized")); return; }
  try {
    const payload = jwt.verify(token, env.JWT_SECRET) as JwtPayload;
    if (typeof payload.sub !== "string") throw new Error("Bad token payload");
    (socket as AuthSocket).userId = payload.sub;
    next();
  } catch {
    next(new Error("Invalid token"));
  }
});

io.on("connection", (socket) => {
  const { userId } = socket as AuthSocket;

  if (!userSockets.has(userId)) userSockets.set(userId, new Set());
  userSockets.get(userId)!.add(socket.id);

  socket.on("join_room",  (roomId: string) => socket.join(roomId));
  socket.on("leave_room", (roomId: string) => socket.leave(roomId));

  socket.on(
    "send_message",
    async (data: { chatRoomId: string; content: string }, callback?: (msg: unknown) => void) => {
      try {
        const room = await prisma.chatRoom.findUnique({ where: { id: data.chatRoomId } });
        if (!room || (room.buyerId !== userId && room.sellerId !== userId)) return;

        const message = await prisma.message.create({
          data:    { chatRoomId: data.chatRoomId, senderId: userId, content: data.content },
          include: { sender: { select: { id: true, username: true, profileImage: true } } },
        });

        io.to(data.chatRoomId).emit("new_message", message);
        callback?.(message);
      } catch (err) {
        console.error("send_message error:", err);
      }
    }
  );

  socket.on("disconnect", () => {
    userSockets.get(userId)?.delete(socket.id);
    if (userSockets.get(userId)?.size === 0) userSockets.delete(userId);
  });
});

// ─── GLOBAL ERROR HANDLER ─────────────────────────────────────────────────────
app.use((err: unknown, _req: Request, res: Response, _next: NextFunction) => {
  if (err instanceof ZodError) {
    const msg = err.issues.map((i) => `${i.path.join(".")}: ${i.message}`).join("; ");
    res.status(400).json({ error: msg });
    return;
  }
  console.error(err);
  res.status(500).json({ error: "Internal server error" });
});

// ─── START ────────────────────────────────────────────────────────────────────
httpServer.listen(parseInt(env.PORT), () => {
  console.log(`🚀 Server running on port ${env.PORT}`);
});

export { app, io, prisma };
